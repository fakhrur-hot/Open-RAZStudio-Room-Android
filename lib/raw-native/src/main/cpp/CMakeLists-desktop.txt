# ============================================================================
#  razbatch — desktop (Windows/Linux/macOS) batch engine
# ============================================================================
#  Builds the SAME C++ core the Android app uses, minus everything that is
#  Android- or preview-specific, as a standalone CLI. This is what makes a
#  preset produce the same pixels on desktop as on the phone.
#
#  DELIBERATELY EXCLUDED (and why):
#    v3_jni.cpp                  JNI entry points — replaced by razbatch_main
#    lens_corrector.cpp          pure JNI glue (3 of its 4 functions are Java_*);
#                                the real lensfun math is lensfun_android.cpp,
#                                which stage_a.cpp calls directly via lfa_*
#    v3/gles_renderer.cpp        GLES preview renderer — batch never uses it
#    v3/offscreen_save_renderer  GLES
#    v3/lut3d.cpp                #includes <GLES3/gl3.h>; only gles_renderer
#                                uses it. Stage C takes the LUT as a raw float
#                                array (StageCOptions::lutData), so the CPU
#                                path needs none of it.
#    v3/stage_b_*.cpp            Stage B is the interactive preview; the batch
#                                path is Stage A -> Stage C directly.
#
#  The batch sources themselves are compiled UNMODIFIED. Their
#  `#include <android/log.h>` resolves to compat/android/log.h, which is on
#  the include path for this target only.
#
#  Configure (Windows, from repo root):
#     cmake -S lib/raw-native/src/main/cpp -B build/razbatch ^
#           -DCMAKE_BUILD_TYPE=Release ^
#           -C lib/raw-native/src/main/cpp/CMakeLists-desktop.txt
#  ...or simply point CMake at this file as the project list file:
#     cmake -S . -B build/razbatch -DCMAKE_BUILD_TYPE=Release
#           (with this file renamed/symlinked to CMakeLists.txt in a side dir)
#  The convenience script tools/build-razbatch.ps1 does this for you.
# ============================================================================

# ── ONNX Runtime (for Zero-DCE inference in AI features) ─────────────────────
# Set ONNXRUNTIME_ROOT environment variable to your ONNX Runtime install dir,
# or download from https://github.com/microsoft/onnxruntime/releases (Windows CPU provider).
# Expected structure:
#   ONNXRUNTIME_ROOT/
#     include/onnxruntime/ (headers)
#     lib/                 (onnxruntime.lib, and onnxruntime.dll on the
#                            official release zip — there is no bin/ dir)
#
# MUST run before razstudio's add_executable()/add_custom_command() below —
# RAZ_HAVE_ONNXRUNTIME gates a POST_BUILD DLL-copy step on that target, and a
# variable used before it's set silently evaluates false in CMake (no error),
# so ordering here is load-bearing, not cosmetic.
if(DEFINED RAZ_ONNXRUNTIME_ROOT)
    set(ONNXRUNTIME_ROOT "${RAZ_ONNXRUNTIME_ROOT}")
elseif(DEFINED ENV{ONNXRUNTIME_ROOT})
    set(ONNXRUNTIME_ROOT "$ENV{ONNXRUNTIME_ROOT}")
else()
    # Fallback: try to find in Program Files or build directory
    if(EXISTS "C:/onnxruntime")
        set(ONNXRUNTIME_ROOT "C:/onnxruntime")
    elseif(EXISTS "${CMAKE_BINARY_DIR}/../onnxruntime")
        set(ONNXRUNTIME_ROOT "${CMAKE_BINARY_DIR}/../onnxruntime")
    else()
        message(WARNING "ONNX Runtime not found. Set ONNXRUNTIME_ROOT env var or download from "
                        "https://github.com/microsoft/onnxruntime/releases")
    endif()
endif()

if(DEFINED ONNXRUNTIME_ROOT AND EXISTS "${ONNXRUNTIME_ROOT}")
    message(STATUS "ONNX Runtime: ${ONNXRUNTIME_ROOT}")
    set(ONNXRUNTIME_INCLUDE_DIR "${ONNXRUNTIME_ROOT}/include")
    set(ONNXRUNTIME_LIB_DIR "${ONNXRUNTIME_ROOT}/lib")
    # The official release zip ships the runtime DLL under lib/, not bin/
    # (there is no bin/ directory at all) — check both to be safe.
    if(EXISTS "${ONNXRUNTIME_ROOT}/bin/onnxruntime.dll")
        set(ONNXRUNTIME_DLL "${ONNXRUNTIME_ROOT}/bin/onnxruntime.dll")
    else()
        set(ONNXRUNTIME_DLL "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll")
    endif()
    set(RAZ_HAVE_ONNXRUNTIME TRUE)
else()
    message(STATUS "ONNX Runtime: NOT FOUND (AI features will be stubbed — "
                   "set ONNXRUNTIME_ROOT and reconfigure to enable them)")
    set(ONNXRUNTIME_INCLUDE_DIR "")
    set(ONNXRUNTIME_LIB_DIR "")
    set(ONNXRUNTIME_DLL "")
    set(RAZ_HAVE_ONNXRUNTIME FALSE)
endif()

    # onnxruntime.dll (+ its shared-providers helper) must sit beside the exe
    # at runtime — a plain target_link_libraries() only satisfies the linker,
    # it does not place the DLL. build-razbatch.ps1 calls `cmake --build`
    # directly (never `cmake --install`), so this has to be a POST_BUILD
    # step rather than an install() rule, or the DLL would silently never
    # get copied and razstudio.exe would fail to launch with "onnxruntime.dll
    # not found".
    if(RAZ_HAVE_ONNXRUNTIME)
        add_custom_command(TARGET razstudio POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${ONNXRUNTIME_DLL}" "$<TARGET_FILE_DIR:razstudio>"
        )
        set(_onnx_shared_dll "${ONNXRUNTIME_ROOT}/lib/onnxruntime_providers_shared.dll")
        if(EXISTS "${_onnx_shared_dll}")
            add_custom_command(TARGET razstudio POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                        "${_onnx_shared_dll}" "$<TARGET_FILE_DIR:razstudio>"
            )
        endif()
    endif()
endif()

    target_include_directories(razparity PRIVATE
            # Parity stubs FIRST: pthread.h + android/{hardware_buffer,
            # native_window}.h shims that let gles_renderer.h (ShaderParams)
            # parse without a real Android sysroot. compat/ still supplies
            # android/log.h. khronos/ supplies EGL/GLES3 headers.
            ${CPP_ROOT}/desktop/parity/compat

    )
    target_compile_definitions(razparity PRIVATE

    endif()

# Same semantics as the Android target: LibRaw without its own threading, and
# the Rayxie vertical CA pass skipped (see CMakeLists.txt for the measurement
# that justified it — keeping them identical preserves output parity).
target_compile_definitions(${_t} PRIVATE
        LIBRAW_NOTHREADS
        RAYXIE_SKIP_VERTICAL
        # Windows only: libraw_types.h declares its API __declspec(dllimport)
        # by default (it assumes you LINK a LibRaw DLL). We compile LibRaw's
        # sources straight into this exe, so without LIBRAW_NODLL every
        # definition fails with C2491 "definition of dllimport function not
        # allowed". Harmless on other platforms — DllDef is empty there anyway.
        LIBRAW_NODLL
        # Skip the EGL/GLES Karis-bloom path in stage_c_export.cpp. The existing
        # `ok == false` branch then runs, which sets karisBloomFallback and lets
        # Orton fall back to Gaussian — the SAME path Android already takes when
        # GPU bloom fails, so this is a pre-existing, logged behaviour rather
        # than a new one. See "Known divergence" in WinBatch/README.md.
        RAZ_NO_EGL
        # libxml2 is built and linked STATICALLY here. Without this its headers
        # declare XMLPUBFUN as __declspec(dllimport) (xmlexports.h:18), and the
        # link fails with "undefined symbol: __declspec(dllimport) xmlParseChunk".
        LIBXML_STATIC
)
endforeach()

# ── Optimisation ────────────────────────────────────────────────────────────
# The Android target forces -O3 even in Debug because the pixel loops are
# unusable at -O0. Mirror that intent per-compiler. -ffp-contract=fast / fp:fast
# matches the Android build's FMA behaviour, which matters for bit-level parity.
#
# TOOLCHAIN REQUIREMENT — clang-cl, not plain MSVC.
# The FP16 pipeline stores pixels as `__fp16` (stage_a, raw_v3_clahe/nr/detail,
# stage_b_downsample, lensfun_android — 10 files). `__fp16` is a Clang/ARM
# extension that cl.exe does NOT implement; building with MSVC fails with
# "C2065: '__fp16': undeclared identifier". clang-cl supports it as a storage
# type on x86-64, so the pipeline compiles unchanged and keeps bit-level
# behaviour. Substituting a struct/uint16 shim was rejected: it would silently
# change rounding and break parity with the phone, which is the one thing this
# port exists to preserve.
#
if(MSVC AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR
        "razbatch requires clang-cl (MSVC lacks __fp16).\n"
        "Install: Visual Studio Installer -> Individual Components ->\n"
        "         'C++ Clang tools for Windows'\n"
        "Then configure with -T ClangCL (build-razbatch.ps1 does this for you).")
endif()

if(MSVC)   # clang-cl: MSVC-style driver, so MSVC-style flags apply
    # NOT /fp:fast. That flag was chosen believing it matched the Android
    # build's "-ffp-contract=fast", but under clang-cl /fp:fast implies full
    # fast-math INCLUDING -ffinite-math-only — which constant-folds
    # std::isnan() to false. lmmse_demosaic.cpp marks unfilled chroma samples
    # with NaN sentinels and fills them via isnan checks; with /fp:fast the
    # fill pass skipped EVERY pixel and the fast-math clamp flushed the NaNs
    # to 0, zeroing R and B at all interpolated sites (75% of pixels) — the
    # strong green cast in every desktop render (whole-image proof: output
    # R/B plane means were EXACTLY the sensel-site means / 4). Android uses
    # only contraction, so the phone was never affected. Match Android
    # precisely: FMA contraction on, errno off, NaN semantics INTACT.
    foreach(_t ${RAZ_TARGETS})
    target_compile_options(${_t} PRIVATE /O2 /W0 /bigobj /utf-8
        /clang:-fno-math-errno /clang:-ffp-contract=fast)
    target_compile_definitions(${_t} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    #
    # F16C: on x86-64 clang lowers __fp16 <-> float conversions to the
    # compiler-rt libcalls __extendhfsf2 / __truncsfhf2. Visual Studio's LLVM
    # component does not ship clang_rt.builtins for x86_64, so those link as
    # "undefined symbol". Enabling F16C makes clang emit the native
    # vcvtph2ps / vcvtps2ph instructions instead — no runtime library needed,
    # and faster.
    #
    # Numerically safe: F16C is IEEE-754 binary16, the same format ARM's
    # native __fp16 uses on the phone, so conversions are bit-identical and
    # parity is preserved. It only enables the conversion instructions; other
    # float math and vectorisation are untouched.
    #
    # Hardware floor: Intel Ivy Bridge (2012) / AMD Piledriver (2012). If you
    # must support older CPUs, drop this and link clang_rt.builtins instead.
    target_compile_options(${_t} PRIVATE /clang:-mf16c)
    endforeach()
else()
    foreach(_t ${RAZ_TARGETS})
        target_compile_options(${_t} PRIVATE -O3 -fno-math-errno -ffp-contract=fast -w)
    endforeach()
endif()
